import java.util.*;
import java.util.stream.Collectors;

public class Main {
    
    static class Node{
        int val;
        Node next;
        Node(int value){
            this.val=value;
            this.next=null;
        }
        
        public static Node addFirst(int value){
           return new Node(value);
          }
          
       public static Node addElement(int value,Node head){
           Node newNode=new Node(value);
           
           if (head==null){
              head=newNode;
              return head;}
           Node temp=head;
           while(temp.next!=null){
               temp=temp.next;
           }
           temp.next=newNode;
           return head;
       }
       public static Node reverseNode(Node head){
           Node prev=null;
           Node curr=head;
           Node temp=head;
           while(curr!=null){
               temp=temp.next;
               curr.next=prev;
               prev=curr;
               curr=temp;
           }
           head=prev;
           return head;
       }
       public static void print(Node head){
           Node temp=head;
           while(temp!=null){
               System.out.println(temp.val);
               temp=temp.next;
           }
            System.out.println("null");
       }
    }
    
    public static void main(String[] args) {
       System.out.println("Enter the value:");
       Scanner sc=new Scanner(System.in);
       int value=sc.nextInt();
       Node head=null;
       head = Node.addFirst(value);
       head = Node.addElement(20, head);
       head = Node.addElement(30, head);
       System.out.println("Original List:");
        Node.print(head);

        head =Node.reverseNode(head);

        System.out.println("Reversed List:");
        Node.print(head);
      
    }
}
